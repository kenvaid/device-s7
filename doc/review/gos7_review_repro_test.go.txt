// Review-only contract tests: injected through Go overlay, never copied into gos7.
type reviewBackend struct{ send func([]byte) ([]byte, error) }

func (b *reviewBackend) Send(p []byte) ([]byte, error) { return b.send(p) }
func (b *reviewBackend) Verify(_, _ []byte) error      { return nil }
func reviewFake(fn func([]byte) ([]byte, error)) *client {
	b := &reviewBackend{send: fn}
	return NewClient2(b, b).(*client)
}
func reviewPacket(n int) []byte {
	p := make([]byte, n)
	p[0], p[4], p[5], p[6], p[7], p[8] = 3, 2, 0xf0, 0x80, 0x32, 3
	binary.BigEndian.PutUint16(p[2:], uint16(n))
	p[11] = 5
	return p
}
func reviewReadReply(data []byte) []byte {
	p := reviewPacket(25 + len(data))
	p[19], p[20], p[21], p[22] = 4, 1, 0xff, tsResByte
	binary.BigEndian.PutUint16(p[13:], 2)
	binary.BigEndian.PutUint16(p[15:], uint16(4+len(data)))
	binary.BigEndian.PutUint16(p[23:], uint16(len(data)*8))
	copy(p[25:], data)
	return p
}
func reviewWriteReply() []byte {
	p := reviewPacket(22)
	p[19], p[20], p[21] = 5, 1, 0xff
	binary.BigEndian.PutUint16(p[13:], 2)
	binary.BigEndian.PutUint16(p[15:], 1)
	return p
}
func reviewReadPacket(c net.Conn) ([]byte, error) {
	h := make([]byte, 4)
	if _, err := io.ReadFull(c, h); err != nil {
		return nil, err
	}
	n := int(binary.BigEndian.Uint16(h[2:]))
	if n < 4 {
		return nil, errors.New("invalid test request length")
	}
	p := make([]byte, n)
	copy(p, h)
	_, err := io.ReadFull(c, p[4:])
	return p, err
}
func reviewOne(t *testing.T, reply []byte) (*client, *TCPClientHandler, <-chan []byte) {
	t.Helper()
	a, b := net.Pipe()
	h := NewTCPClientHandler("127.0.0.1:102", 0, 1)
	h.conn, h.PDULength, h.IdleTimeout, h.Timeout = a, 480, 0, time.Second
	requests := make(chan []byte, 1)
	go func() {
		defer b.Close()
		if p, err := reviewReadPacket(b); err == nil {
			requests <- p
			_, _ = b.Write(reply)
		}
	}()
	t.Cleanup(func() { _ = h.Close(); _ = b.Close() })
	return NewClient(h).(*client), h, requests
}
func reviewNoPanic(t *testing.T) {
	t.Helper()
	if p := recover(); p != nil {
		t.Errorf("must return an error, panicked: %v", p)
	}
}
func reviewItem(word, amount int, data []byte) []S7DataItem {
	area, db := s7areadb, 1
	if word == s7wlcounter {
		area, db = s7areact, 0
	} else if word == s7wltimer {
		area, db = s7areatm, 0
	}
	return []S7DataItem{{Area: area, WordLen: word, DBNumber: db, Start: 10, Bit: 3, Amount: amount, Data: data}}
}
func reviewSZL(data []byte, first, done bool) []byte {
	offset, extra := 37, 0
	if first {
		offset, extra = 41, 8
	}
	p := reviewPacket(offset + len(data))
	p[8], p[24], p[29], p[30] = 7, 1, 0xff, tsResOctet
	if !done {
		p[26] = 1
	}
	binary.BigEndian.PutUint16(p[31:], uint16(len(data)+extra))
	if first {
		binary.BigEndian.PutUint16(p[37:], 2)
		binary.BigEndian.PutUint16(p[39:], 2)
	}
	copy(p[offset:], data)
	return p
}

func TestReviewGos7Contracts(t *testing.T) {
	t.Run("bit_read_write_same_address", func(t *testing.T) {
		read, _, rq := reviewOne(t, reviewReadReply([]byte{1}))
		items := reviewItem(s7wlbit, 1, []byte{1})
		if err := read.AGReadMulti(items, 1); err != nil {
			t.Fatal(err)
		}
		rp := <-rq
		write, _, wq := reviewOne(t, reviewWriteReply())
		if err := write.AGWriteMulti(items, 1); err != nil {
			t.Fatal(err)
		}
		wp := <-wq
		if !bytes.Equal(rp[28:31], wp[28:31]) {
			t.Errorf("same Start=10 Bit=3: read address=%x write address=%x; expected 000053", rp[28:31], wp[28:31])
		}
	})
	t.Run("timeout_discards_connection_and_stale_reply", func(t *testing.T) {
		a, b := net.Pipe()
		h := NewTCPClientHandler("127.0.0.1:102", 0, 1)
		h.conn, h.PDULength, h.Timeout, h.IdleTimeout = a, 480, 30*time.Millisecond, 0
		t.Cleanup(func() { _ = h.Close(); _ = b.Close() })
		firstTimedOut := make(chan struct{})
		go func() {
			defer b.Close()
			if _, err := reviewReadPacket(b); err != nil {
				return
			}
			<-firstTimedOut
			if _, err := reviewReadPacket(b); err != nil {
				return
			}
			// Deliver the first request's late response after receiving the second.
			_, _ = b.Write(reviewReadReply([]byte{0x11}))
		}()
		c := NewClient(h)
		if err := c.AGReadDB(1, 0, 1, make([]byte, 1)); err == nil {
			t.Fatal("expected first read timeout")
		}
		h.mu.Lock()
		retained := h.conn != nil
		h.mu.Unlock()
		if retained {
			t.Error("timed-out connection retained")
		}
		close(firstTimedOut)
		data := []byte{0}
		if err := c.AGReadDB(1, 1, 1, data); err == nil {
			t.Errorf("second address read accepted stale first response: %x", data)
		}
	})
	t.Run("global_read_error_propagated", func(t *testing.T) {
		p := reviewReadReply([]byte{0x42})
		p[17], p[18] = 0x81, 0x04
		c, _, _ := reviewOne(t, p)
		if err := c.AGReadDB(1, 0, 1, make([]byte, 1)); err == nil {
			t.Error("S7 Ack_Data error 8104 ignored")
		}
	})
	t.Run("global_write_error_propagated", func(t *testing.T) {
		p := reviewWriteReply()
		p[17], p[18] = 0x81, 0x04
		c, _, _ := reviewOne(t, p)
		if err := c.AGWriteDB(1, 0, 1, []byte{1}); err == nil {
			t.Error("S7 Ack_Data error 8104 ignored")
		}
	})
	for _, kind := range []string{"wrong_reference", "wrong_function", "wrong_protocol", "wrong_tpkt_version", "wrong_cotp_type"} {
		t.Run(kind, func(t *testing.T) {
			p := reviewReadReply([]byte{0x42})
			switch kind {
			case "wrong_reference":
				p[11] = 0xaa
			case "wrong_function":
				p[19] = 5

			if err := c.AGReadDB(1, 0, 1, make([]byte, 1)); err == nil {
				t.Error("mismatched response accepted")
			}
		})
	}
	t.Run("single_read_rejects_short_payload", func(t *testing.T) {

func TestReviewGos7ConcurrentIO(t *testing.T) {
	a, b := net.Pipe()
	h := NewTCPClientHandler("127.0.0.1:102", 0, 1)
	h.conn, h.PDULength, h.IdleTimeout, h.Timeout = a, 480, 0, time.Second
	defer h.Close()
	defer b.Close()
	go func() {
		for {
			p, err := reviewReadPacket(b)
			if err != nil {
				return
			}
			var reply []byte
			if p[17] == 5 {
				reply = reviewWriteReply()
			} else {
				reply = reviewReadReply([]byte{byte(p[30] / 8)})
			}
			if _, err := b.Write(reply); err != nil {
				return
			}
		}
	}()
	c := NewClient(h)
	var wg sync.WaitGroup
	for w := 0; w < 4; w++ {
		wg.Add(1)
		go func(start int) {
			defer wg.Done()
			for i := 0; i < 50; i++ {
				data := []byte{0}
				if err := c.AGReadDB(1, start, 1, data); err != nil || data[0] != byte(start) {
					t.Errorf("concurrent read %d: %x %v", start, data, err)
					return
				}
				if err := c.AGWriteDB(1, start, 1, []byte{1}); err != nil {
					t.Error(err)
					return
				}
			}
		}(w)
	}
	wg.Wait()
}

	defer b.Close()
	go func() {
		for {
			p, err := reviewReadPacket(b)
			if err != nil {
				return
			}
			var reply []byte
			if p[5] == 0xe0 {
				reply = reviewPacket(22)
				reply[5] = 0xd0
			} else if p[17] == 0xf0 {
				reply = reviewPacket(27)
				binary.BigEndian.PutUint16(reply[25:], 480)
			} else {
				reply = reviewReadReply([]byte{1})
			}
			if _, err := b.Write(reply); err != nil {
				return
			}
		}
	}()
	c := NewClient(h)
	var wg sync.WaitGroup
	start := make(chan struct{})
	wg.Add(2)
	go func() {
		defer wg.Done()
		<-start
		for i := 0; i < 200; i++ {
			if err := h.ConnectContext(context.Background()); err != nil {
				t.Logf("concurrent connect: %v", err)
				return
			}
		}
	}()
	go func() {
		defer wg.Done()
		<-start
		for i := 0; i < 500; i++ {
			if err := c.AGReadDB(1, 0, 1, make([]byte, 1)); err != nil {
				t.Logf("concurrent read: %v", err)
				return
			}
		}
	}()
	close(start)
	wg.Wait()
}
